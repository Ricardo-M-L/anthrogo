# anthrogo M3 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Land MCP support: spawn `mcpServers` from settings.yaml as stdio subprocesses, discover their tools via `tools/list`, expose each as `mcp__<server>__<tool>` in the existing tool registry so the model can call them through the unchanged turn loop + permission gate. Add `/mcp` and `/mcp reload` slash commands.

**Architecture:** A new `internal/mcp` package wraps `github.com/modelcontextprotocol/go-sdk` (v1.6.0 official). `Manager` owns N `Server` instances (one subprocess each), `MCPAdapter` in `pkg/tool` wraps each MCP tool descriptor into the existing `tool.Tool` interface. `cmd/anthrogo` builds the Manager at startup, blocks on all servers reaching ready-or-timeout, merges adapter tools into the registry. No changes to engine, gate, or session storage.

**Tech Stack:** Go 1.23 stdlib + `github.com/modelcontextprotocol/go-sdk@v1.6.0`. No other new deps.

**Reference spec:** `docs/superpowers/specs/2026-05-17-anthrogo-m3-design.md`

> **Critical SDK note:** The exact API surface of `modelcontextprotocol/go-sdk` may differ from what this plan assumes. Every task that touches `sdk.*` MUST start by running `go doc github.com/modelcontextprotocol/go-sdk` and adapting the symbol names. The plan's code blocks show the *target shape* — adjust mechanically when the SDK reality differs. If the SDK doesn't expose a needed primitive, report BLOCKED with what's missing.

package tool

import (
	"crypto/sha256"
	"encoding/hex"
	"fmt"
)

const (
	mcpPrefix         = "mcp__"
	mcpSep            = "__"
	mcpMaxToolNameLen = 64
	mcpServerMaxLen   = 16
	mcpToolMaxLen     = 41 // 64 - len("mcp__") - len("__") - mcpServerMaxLen
)

// MCPToolName composes an MCP tool name from a server name and tool name,
// fitting the Anthropic 64-char limit. When server or tool exceeds its
// per-piece budget, both are truncated and an 8-char sha256 hex suffix is
// appended for collision resistance.
func MCPToolName(server, tool string) string {
	// Fast path: everything fits.
	candidate := mcpPrefix + server + mcpSep + tool
	if len(candidate) <= mcpMaxToolNameLen &&
		len(server) <= mcpServerMaxLen &&
		len(tool) <= mcpToolMaxLen {
		return candidate
	}

	srv := server
	if len(srv) > mcpServerMaxLen {
		srv = srv[:mcpServerMaxLen]
	}
	// Available budget for tool slot including the sha suffix
	const suffixLen = 1 + 8 // "_<hex>" = 9 chars
	// 5 + len(srv) + 2 + len(toolBudget) + 9 ≤ 64
	toolBudget := mcpMaxToolNameLen - len(mcpPrefix) - len(srv) - len(mcpSep) - suffixLen
	if toolBudget < 1 {
		toolBudget = 1
	}
	toolSlice := tool
	if len(toolSlice) > toolBudget {
		toolSlice = toolSlice[:toolBudget]
	}
	sum := sha256.Sum256([]byte(server + "/" + tool))
	suffix := hex.EncodeToString(sum[:4]) // 8 hex chars
	out := fmt.Sprintf("%s%s%s%s_%s", mcpPrefix, srv, mcpSep, toolSlice, suffix)
	if len(out) > mcpMaxToolNameLen {
		// Belt-and-braces hard truncation; should not trigger with budget math above.
		out = out[:mcpMaxToolNameLen]
	}
	return out
}
```

type Server struct {

	Name string
	cfg  MCPServerConfig

	cmd     *exec.Cmd
	session *sdk.ClientSession // type name adapted from sdk surface
	tools   []*sdk.Tool        // result of tools/list

	err   error

// NewServer constructs (but does not start) a Server.
func NewServer(name string, cfg MCPServerConfig, notifyLog func(string, string)) *Server {
	if cfg.Timeout == 0 {
		cfg.Timeout = DefaultInitTimeout
	}
	return &Server{
		Name:      name,
		cfg:       cfg,
		state:     StateInit,
		notifyLog: notifyLog,

// Start spawns the subprocess, performs the initialize handshake, and calls
// tools/list. Returns the first error encountered. On failure, the Server's
// state is StateFailed; tools is nil.
func (s *Server) Start(ctx context.Context) error {
	startCtx, cancel := context.WithTimeout(ctx, s.cfg.Timeout)
	defer cancel()

	cmd := exec.CommandContext(startCtx, s.cfg.Command, s.cfg.Args...)
	cmd.Dir = s.cfg.Cwd
	cmd.Env = mergeEnv(s.cfg.Env)
	// Put the child in its own process group so we can SIGTERM the group on Close.
	cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}

	s.mu.Lock()
	s.cmd = cmd
	s.mu.Unlock()

func (s *Server) fail(err error) {
	s.mu.Lock()
	s.state = StateFailed
	s.err = err
	s.mu.Unlock()
}

// State returns the current state. Safe for concurrent reads.
func (s *Server) State() State {
	s.mu.RLock()
	defer s.mu.RUnlock()
	return s.state
}

// Err returns the last error if state is Failed.
func (s *Server) Err() error {
	s.mu.RLock()
	defer s.mu.RUnlock()
	return s.err
}

// Tools returns the most recent tools/list snapshot.
func (s *Server) Tools() []*sdk.Tool {
	s.mu.RLock()
	defer s.mu.RUnlock()
	out := make([]*sdk.Tool, len(s.tools))
	copy(out, s.tools)
	return out
}

// CallTool invokes the server's tool by name with the given args.
func (s *Server) CallTool(ctx context.Context, name string, args map[string]any) (*sdk.CallToolResult, error) {
	s.mu.RLock()
	sess := s.session
	state := s.state
	s.mu.RUnlock()
	if state != StateReady {
		return nil, fmt.Errorf("mcp server %s is not ready (state=%s)", s.Name, state)
	}
	if sess == nil {
		return nil, errors.New("mcp session not established")
	}
	return sdkCallTool(ctx, sess, name, args)
}

// Close gracefully terminates the server: SIGTERM the process group,
// wait up to 2s, then SIGKILL.
func (s *Server) Close() error {
	s.mu.Lock()
	if s.state == StateClosed {
		s.mu.Unlock()
		return nil
	}
	sess := s.session
	cmd := s.cmd
	s.state = StateClosed
	s.mu.Unlock()

	if sess != nil {
		_ = sess.Close()
	}
	if cmd != nil && cmd.Process != nil {
		// Signal the whole group; if Setpgid wasn't honoured, fall back to PID.
		pgid, err := syscall.Getpgid(cmd.Process.Pid)
		if err == nil {
			_ = syscall.Kill(-pgid, syscall.SIGTERM)
		} else {
			_ = cmd.Process.Signal(syscall.SIGTERM)
		}
		done := make(chan error, 1)
		go func() { done <- cmd.Wait() }()
		select {
		case <-done:
		case <-time.After(2 * time.Second):
			_ = cmd.Process.Kill()
			<-done
		}
	}
	return nil
}

package tool

import (
	"context"
	"encoding/json"
	"fmt"

	sdk "github.com/modelcontextprotocol/go-sdk/mcp"
)

// MCPInvoker is the call-back the manager hands the adapter so the adapter
// doesn't need to import internal/mcp.
type MCPInvoker func(ctx context.Context, args map[string]any) (*sdk.CallToolResult, error)

// MCPAdapter wraps one MCP server's tool descriptor as a tool.Tool.
type MCPAdapter struct {
	DefaultPermission

	composedName string             // mcp__<server>__<tool>
	originalName string             // <tool> as advertised by the server
	server       string
	descriptor   *sdk.Tool
	invoker      MCPInvoker
}

func NewMCPAdapter(serverName string, descriptor *sdk.Tool, invoker MCPInvoker) *MCPAdapter {
	return &MCPAdapter{
		composedName: MCPToolName(serverName, descriptor.Name),
		originalName: descriptor.Name,
		server:       serverName,
		descriptor:   descriptor,
		invoker:      invoker,
	}
}

func (a *MCPAdapter) Name() string                         { return a.composedName }
func (a *MCPAdapter) Description(context.Context) string   {
	return fmt.Sprintf("[%s] %s", a.server, a.descriptor.Description)
}
func (a *MCPAdapter) UserFacingName(map[string]any) string { return a.composedName }
func (a *MCPAdapter) IsReadOnly() bool                     { return false }
func (a *MCPAdapter) IsConcurrencySafe() bool              { return true }

func (a *MCPAdapter) Schema() map[string]any {
	// descriptor.InputSchema is the JSON Schema as the server advertised it.
	// Pass it through. If the SDK exposes it as a Go struct rather than a
	// raw map, marshal-then-unmarshal to coerce.
	if a.descriptor.InputSchema == nil {
		return map[string]any{"type": "object", "properties": map[string]any{}}
	}
	// Try direct assertion first.
	if m, ok := any(a.descriptor.InputSchema).(map[string]any); ok {
		return m
	}
	raw, err := json.Marshal(a.descriptor.InputSchema)
	if err != nil {
		return map[string]any{"type": "object", "properties": map[string]any{}}
	}
	var out map[string]any
	if err := json.Unmarshal(raw, &out); err != nil {
		return map[string]any{"type": "object", "properties": map[string]any{}}
	}
	return out
}

func (a *MCPAdapter) Call(ctx context.Context, input map[string]any, _ *Context) (Result, error) {
	res, err := a.invoker(ctx, input)
	if err != nil {
		msg := fmt.Sprintf("mcp call %s failed: %s", a.composedName, err.Error())
		return Result{Type: ResultText, Text: msg, ForLLM: msg, IsError: true}, nil
	}
	// Flatten the content array: most MCP tool results are 1-N text blocks.
	text := mcpContentToText(res.Content)
	return Result{
		Type:    ResultText,
		Text:    text,
		ForLLM:  text,
		IsError: res.IsError,
		Data:    map[string]any{"server": a.server, "tool": a.originalName},
	}, nil
}

// mcpContentToText joins MCP content blocks into a single string. Image and
// non-text blocks are described as placeholders.
func mcpContentToText(content []sdk.Content) string {
	var parts []string
	for _, c := range content {
		switch v := c.(type) {
		case *sdk.TextContent:
			parts = append(parts, v.Text)
		case *sdk.ImageContent:
			parts = append(parts, fmt.Sprintf("[image: %s]", v.MimeType))
		default:
			raw, _ := json.Marshal(c)
			parts = append(parts, string(raw))
		}
	}
	return joinTrim(parts)
}

func joinTrim(parts []string) string {
	out := ""
	for i, p := range parts {
		if i > 0 {
			out += "\n"
		}
		out += p
	}
	return out
}
```

package mcp

import (
	"context"
	"sync"

	sdk "github.com/modelcontextprotocol/go-sdk/mcp"

	"github.com/ricardo/anthrogo/pkg/tool"
)

// Manager owns N MCP Servers and exposes their tools through tool.Tool
// adapters. Construct with NewManager; call Start before AllTools.
type Manager struct {
	mu      sync.RWMutex
	servers map[string]*Server
	logSink LogSink
}

func NewManager(logSink LogSink) *Manager {
	return &Manager{
		servers: map[string]*Server{},
		logSink: logSink,
	}
}

// AddServer registers a server config. Server is not started until Start().
func (m *Manager) AddServer(name string, cfg MCPServerConfig) {
	m.mu.Lock()
	defer m.mu.Unlock()
	m.servers[name] = NewServer(name, cfg, m.logSink)
}

// Start spawns every server in parallel and waits for all of them to settle
// (ready or failed). Returns nil even if some servers failed — inspect with
// State(name).
func (m *Manager) Start(ctx context.Context) error {
	m.mu.RLock()
	servers := make([]*Server, 0, len(m.servers))
	for _, s := range m.servers {
		servers = append(servers, s)
	}
	m.mu.RUnlock()

	var wg sync.WaitGroup
	for _, s := range servers {
		wg.Add(1)
		go func(srv *Server) {
			defer wg.Done()
			_ = srv.Start(ctx)
		}(s)
	}
	wg.Wait()
	return nil
}

// Names returns server names in deterministic (sorted) order.
func (m *Manager) Names() []string {
	m.mu.RLock()
	defer m.mu.RUnlock()
	out := make([]string, 0, len(m.servers))
	for n := range m.servers {
		out = append(out, n)
	}
	// sort for stable tool ordering across runs (matters for prompt caching).
	sortStrings(out)
	return out
}

// State returns the state of one server, or "" if no such name.
func (m *Manager) State(name string) State {
	m.mu.RLock()
	defer m.mu.RUnlock()
	s, ok := m.servers[name]
	if !ok {
		return ""
	}
	return s.State()
}

// Err returns the last error for one server.
func (m *Manager) Err(name string) error {
	m.mu.RLock()
	defer m.mu.RUnlock()
	s, ok := m.servers[name]
	if !ok {
		return nil
	}
	return s.Err()
}

// AllTools returns one MCPAdapter per (ready server, advertised tool).
func (m *Manager) AllTools() []tool.Tool {
	m.mu.RLock()
	defer m.mu.RUnlock()
	var out []tool.Tool
	for _, name := range m.namesLocked() {
		s := m.servers[name]
		if s.State() != StateReady {
			continue
		}
		for _, t := range s.Tools() {
			descriptor := t
			invoker := func(ctx context.Context, args map[string]any) (*sdk.CallToolResult, error) {
				return s.CallTool(ctx, descriptor.Name, args)
			}
			out = append(out, tool.NewMCPAdapter(name, descriptor, invoker))
		}
	}
	return out
}

func (m *Manager) namesLocked() []string {
	out := make([]string, 0, len(m.servers))
	for n := range m.servers {
		out = append(out, n)
	}
	sortStrings(out)
	return out
}

// Close terminates every server.
func (m *Manager) Close() error {
	m.mu.Lock()
	defer m.mu.Unlock()
	for _, s := range m.servers {
		_ = s.Close()
	}
	return nil
}

// sortStrings inlines sort.Strings to avoid importing "sort" if the file
// stays minimal. (If you prefer, just `import "sort"` and use sort.Strings.)
func sortStrings(s []string) {
	for i := 1; i < len(s); i++ {
		for j := i; j > 0 && s[j-1] > s[j]; j-- {
			s[j-1], s[j] = s[j], s[j-1]
		}
	}
}

func (MCP) Type() command.Type  { return command.TypeLocal }

func (MCP) Run(ctx context.Context, args string, host command.Host) (command.Result, error) {
	mgr := host.MCP()
	if mgr == nil {
		return command.Result{Text: "no MCP manager configured"}, nil
	}
	args = strings.TrimSpace(args)
	switch {
	case args == "":
		return listServers(ctx, mgr, host), nil
	case args == "reload":
		// Close + re-start every server. The Manager's existing config map is reused.
		_ = mgr.Close()
		if err := mgr.Start(ctx); err != nil {
			return command.Result{}, err
		}
		return command.Result{Text: "MCP servers reloaded"}, nil
	case strings.HasPrefix(args, "status "):
		name := strings.TrimSpace(strings.TrimPrefix(args, "status"))
		return command.Result{Text: statusDetail(mgr, name)}, nil
	default:
		return command.Result{Text: "usage: /mcp [reload | status <name>]"}, nil
	}
}

func listServers(_ context.Context, mgr mcpStatus, host command.Host) command.Result {
	names := mgr.Names()
	if len(names) == 0 {
		return command.Result{Text: "(no MCP servers configured)"}
	}
	var b strings.Builder
	b.WriteString("MCP servers:\n")
	for _, n := range names {
		state := mgr.State(n)
		toolCount := 0
		for _, t := range host.Tools().All() {
			if strings.HasPrefix(t.Name(), "mcp__"+n+"__") {
				toolCount++
			}
		}
		fmt.Fprintf(&b, "  %s  state=%s  tools=%d\n", n, state, toolCount)
	}
	return command.Result{Text: b.String()}
}

func statusDetail(mgr mcpStatus, name string) string {
	state := mgr.State(name)
	if state == "" {
		return "no such MCP server: " + name
	}
	err := mgr.Err(name)
	out := fmt.Sprintf("server: %s\nstate: %s\n", name, state)
	if err != nil {
		out += "last error: " + err.Error() + "\n"
	}
	return out
}
